async function notifyOffice() { const qrId = document.getElementById("qrId").value.trim(); if (!qrId) { alert("Please enter QR ID."); return; } try { // Find the vehicle using QR_ID const { data: vehicle, error: vehicleError } = await db .from("cars") .select("QR_ID, CAR_NUMBER") .eq("QR_ID", qrId) .maybeSingle(); if (vehicleError) { console.error("Vehicle lookup error:", vehicleError); alert("Vehicle lookup failed: " + vehicleError.message); return; } if (!vehicle) { alert("QR ID not found."); return; } // Create notification const { error: notificationError } = await db .from("notifications") .insert({ qr_id: vehicle.QR_ID, car_number: vehicle.CAR_NUMBER, message: "Customer has requested office assistance.", status: "NEW" }); if (notificationError) { console.error( "Notification insert error:", notificationError ); alert( "Notification failed: " + notificationError.message ); return; } alert("✅ Office has been notified!"); } catch (error) { console.error("Unexpected error:", error); alert( "Unexpected error: " + error.message ); } }
